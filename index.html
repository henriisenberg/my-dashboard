}
}
};

}

}

}
paint(item);
}
}
return placed;
}

function rssImage(item) {
const nodes = item.getElementsByTagName('*');
for (const el of nodes) {
const name = (el.localName || el.nodeName || '').toLowerCase();
if (!/content|thumbnail|enclosure/.test(name)) continue;
const url = el.getAttribute('url') || el.getAttribute('href') || '';
if (/^https?:/i.test(url) && !/\.(mp3|mp4|m4a)(\?|$)/i.test(url)) return url;
}
const blob = [...nodes].map(el => el.textContent || '').join('\n');
const match = blob.match(/<img[^>]+src=["']([^"']+)["']/i);
return match ? match[1] : '';
}

async function fetchRSS(url) {
// Skip known-bad Google News RSS feeds
if (url.includes('news.google.com')) {
return [];
}

const proxies = [PROXY, 'https://corsproxy.io/?'];
for (const proxy of proxies) {
try {
const target = proxy.endsWith('?') ? proxy + encodeURIComponent(url) : proxy + url;
const res = await fetch(target, { credentials: 'omit' });
if (!res.ok) {
if (res.status === 503) continue;
throw new Error(res.status);
}
const text = await res.text();
const parser = new DOMParser();
const xml = parser.parseFromString(text, 'text/xml');
const items = [...xml.querySelectorAll('item')].slice(0, 12).map(item => {
const image = rssImage(item);
const desc = item.querySelector('description')?.textContent || '';
const host = (() => { try { return new URL(url).hostname; } catch (e) { return url; } })();
const source = host.includes('fox') ? 'Fox' : host.includes('cnn') ? 'CNN' : host.includes('reuters') ? 'Reuters' : host.includes('bbc') ? 'BBC' : host.includes('latimes') ? 'LA Times' : host.includes('nytimes') ? 'NY Times' : host.includes('axios') ? 'Axios' : host.includes('npr') ? 'NPR' : host.includes('timesofisrael') ? 'Times of Israel' : 'News';
return {
title: item.querySelector('title')?.textContent || '',
link: item.querySelector('link')?.textContent || '',
pubDate: item.querySelector('pubDate')?.textContent || '',
image,
summary: desc.replace(/<[^>]*>/g, '').slice(0, 140).trim(),
source
};
});
if (items.length) return items;
} catch (e) {}
}
return [];
}

const fitNudge = { news: 0, local: 0, intl: 0, sports: 0, stocks: 0, tech: 0, clocks: 0 };

function dotsHtml(total, visible, start) {
if (total <= visible) return '';
const pages = Math.max(1, Math.ceil(total / visible));
const current = Math.floor(start / visible) % pages;
const dots = Array.from({ length: pages }, (_, i) =>
`<span class="dot${i === current ? ' on' : ''}"></span>`).join('');
return `<div class="rot-dots">${dots}</div>`;
}

function howMany(el, itemH, key) {
const h = el.clientHeight;
if (h < 40) return 0;
const usable = Math.max(itemH, h - 22);
const base = Math.max(1, Math.floor(usable / itemH));
return Math.max(1, base - (fitNudge[key] || 0));
}

function tightenIfOverflow(el, key, render) {
const scroller = el.querySelector('.items-scroller');
if (!scroller) return;
if (scroller.scrollHeight > scroller.clientHeight + 6 && (fitNudge[key] || 0) < 8) {
fitNudge[key] = (fitNudge[key] || 0) + 1;
render();
}
}

function gameHasFavorite(game, fav) {
if (!fav || game.leagueKey !== fav.league) return false;
const abbr = normAbbr(fav.league, fav.abbr);
const away = normAbbr(game.leagueKey, game.away);
const home = normAbbr(game.leagueKey, game.home);
if (abbr && (abbr === away || abbr === home)) return true;
const name = (fav.name || '').toLowerCase();
if (!name) return false;
return (game.awayName || '').toLowerCase().includes(name) || (game.homeName || '').toLowerCase().includes(name);
}

async function nextFavoriteGame(fav, games) {
const mine = games.filter(g => gameHasFavorite(g, fav));
if (mine.some(g => g.isLive || g.state === 'pre')) return null;

});
}

function orderSports(games) {

if (a.isLive !== b.isLive) return a.isLive ? -1 : 1;
if (a.isUpcoming !== b.isUpcoming) return a.isUpcoming ? -1 : 1;
return (Date.parse(a.date) || 0) - (Date.parse(b.date) || 0);
});
const restOn = (g) => (settings.restLeagues || []).includes(g.leagueKey) && (settings.sportsLeagues || []).includes(g.leagueKey);
const rest = games.filter(g => !g.favorite && restOn(g)).sort((a, b) => (a.isLive === b.isLive ? 0 : a.isLive ? -1 : 1));
return favorites.concat(rest);
}

const res = await fetch(PROXY + url);
if (!res.ok) throw new Error(res.status);

if (!result) return null;

const price = meta.regularMarketPrice;

function zoneParts(tz, date) {
const parts = Object.fromEntries(new Intl.DateTimeFormat('en-US', {
timeZone: tz, hour: '2-digit', minute: '2-digit', second: '2-digit', weekday: 'short', hourCycle: 'h23'
}).formatToParts(date).map(p => [p.type, p.value]));
return { h: +parts.hour, m: +parts.minute, s: +parts.second, day: parts.weekday };
}

function solarMinutes(lat, lon, tz) {
const now = new Date();
const parts = zoneParts(tz, now);
const utc = zoneParts('UTC', now);
const local = parts.h * 60 + parts.m + parts.s / 60;
let offset = (parts.h * 60 + parts.m) - (utc.h * 60 + utc.m);
if (offset > 720) offset -= 1440;
if (offset < -720) offset += 1440;
const day = Object.fromEntries(new Intl.DateTimeFormat('en-US', { timeZone: tz, year:'numeric', month:'2-digit', day:'2-digit' }).formatToParts(now).map(p => [p.type, p.value]));
const y = +day.year, m = +day.month, d = +day.day;
const doy = Math.floor((Date.UTC(y, m - 1, d) - Date.UTC(y, 0, 0)) / 86400000);
const lngHour = lon / 15;
const calc = (isRise) => {
const t = doy + ((isRise ? 6 : 18) - lngHour) / 24;
const M = (0.9856 * t) - 3.289;
let L = M + (1.916 * Math.sin(M * Math.PI / 180)) + (0.020 * Math.sin(2 * M * Math.PI / 180)) + 282.634;
L = ((L % 360) + 360) % 360;
let RA = Math.atan(0.91764 * Math.tan(L * Math.PI / 180)) * 180 / Math.PI;
RA = ((RA % 360) + 360) % 360;
RA = (RA + (Math.floor(L / 90) * 90 - Math.floor(RA / 90) * 90)) / 15;
const sinDec = 0.39782 * Math.sin(L * Math.PI / 180);
const cosDec = Math.cos(Math.asin(sinDec));
const cosH = (Math.cos(90.833 * Math.PI / 180) - sinDec * Math.sin(lat * Math.PI / 180)) / (cosDec * Math.cos(lat * Math.PI / 180));
if (cosH > 1 || cosH < -1) return null;
let H = (isRise ? 360 - Math.acos(cosH) * 180 / Math.PI : Math.acos(cosH) * 180 / Math.PI) / 15;
let UT = (H + RA - (0.06571 * t) - 6.622 - lngHour) % 24;
if (UT < 0) UT += 24;
let localH = UT + offset / 60;
if (localH < 0) localH += 24;
if (localH >= 24) localH -= 24;
return localH * 60;
};
let rise = calc(true);
let set = calc(false);
if (rise != null && set != null && set < rise) set += 1440;
return { local, rise, set };
}

function fmtMin(mins) {
if (mins == null || Number.isNaN(mins)) return '—';
const m = ((Math.round(mins) % 1440) + 1440) % 1440;
const h24 = Math.floor(m / 60);
const mm = String(m % 60).padStart(2, '0');
const h12 = h24 % 12 || 12;
return `${h12}:${mm} ${h24 < 12 ? 'AM' : 'PM'}`;
}

function sunArc(sun, id) {
const left = 12, right = 288, base = 168, amp = 112;
const rise = sun.rise;
const set = sun.set > 1440 ? sun.set - 1440 : sun.set;
const daySpan = Math.max(1, set - rise);
const nightSpan = Math.max(1, 1440 - daySpan);
const local = ((sun.local % 1440) + 1440) % 1440;
const night = local < rise || local > set;
let t;
if (!night) t = (local - rise) / daySpan;
else if (local > set) t = (local - set) / nightSpan;
else t = (local + 1440 - set) / nightSpan;
t = Math.max(0, Math.min(1, t));
const pts = [];
for (let i = 0; i <= 36; i++) {
const u = i / 36;
pts.push([left + u * (right - left), base - Math.sin(u * Math.PI) * amp]);
}
const d = pts.map((p, i) => `${i ? 'L' : 'M'}${p[0].toFixed(1)} ${p[1].toFixed(1)}`).join(' ');
const sx = left + t * (right - left);
const sy = base - Math.sin(t * Math.PI) * amp;
const skyTop = night ? '#121a2b' : '#1d4e89';
const skyBot = night ? '#0b1020' : '#12345f';
const marker = `<div class="sun-marker${night ? ' moon' : ''}" style="left:${(sx / 300 * 100).toFixed(2)}%;top:${(sy / 180 * 100).toFixed(2)}%"></div>`;
const leftName = night ? 'Sunset' : 'Sunrise';
const rightName = night ? 'Sunrise' : 'Sunset';
const leftWhen = fmtMin(night ? set : rise);
const rightWhen = fmtMin(night ? rise : set);
const svg = `<svg class="sun-arc" viewBox="0 0 300 180" preserveAspectRatio="none" role="img" aria-label="${leftName} ${leftWhen}, ${rightName} ${rightWhen}">
<defs>
<linearGradient id="sky${id}" x1="0" y1="0" x2="0" y2="1">
<stop offset="0" stop-color="${skyTop}"/>
<stop offset="1" stop-color="${skyBot}"/>
</linearGradient>
</defs>
<rect width="300" height="180" fill="url(#sky${id})"/>
<path d="${d}" fill="none" stroke="${night ? 'rgba(226,232,240,0.85)' : 'rgba(248,250,252,0.95)'}" stroke-width="2.2" stroke-linecap="round" vector-effect="non-scaling-stroke"/>
</svg>`;
const times = `<div class="sun-times"><span><em>${leftName}</em>${leftWhen}</span><span style="text-align:right"><em>${rightName}</em>${rightWhen}</span></div>`;
return { svg, times, marker, night };
}

function clockFace(raw, compact) {
const c = cityCoords(raw);
let parts = { h: 0, m: 0, s: 0, day: '' };
let digital = '—';
let dateLine = '';
try {
parts = zoneParts(c.tz, new Date());
digital = new Date().toLocaleTimeString('en-US', { timeZone: c.tz, hour: 'numeric', minute: '2-digit' });
dateLine = new Date().toLocaleDateString('en-US', { timeZone: c.tz, weekday: 'short', month: 'short', day: 'numeric' });
} catch (e) {}
const sun = (c.lat != null) ? solarMinutes(c.lat, c.lon, c.tz) : null;
const gid = String(c.tz || c.label || 'c').replace(/\W/g, '');
const sunUi = (sun && sun.rise != null && sun.set != null) ? sunArc(sun, gid) : { svg: '', times: '', marker: '', night: false };
const tone = sunUi.svg ? (sunUi.night ? 'night' : 'day') : '';
return `<div class="clock-card ${tone}"><div class="sun-stage">${sunUi.svg || ''}${sunUi.marker || ''}</div><div class="clock-head"><div class="clock-meta"><div class="meta">${c.label}</div>${compact ? '' : `<div class="meta">${dateLine}</div>`}</div>${analogSVG(parts.h, parts.m, parts.s)}</div><div class="clock-time">${digital}</div>${sunUi.times || ''}</div>`;
}
